## 二、当前配置详情

### 配置文件: `expconfigs/distill.yaml`

```yaml
run:
  exp_name: "K_only_student_v1"
  seed: 42
  max_epochs: 30
  early_stop_patience: 10  # ← 已修复（原为3）
  ckpt_interval: 1

data:
  train_set: "datasets/train_pairs.lst"  # 998对真实图像
  val_set: "datasets/val_pairs.lst"      # 998对真实图像
  num_workers: 8
  batch_size: 1

teacher:
  weights: "checkpoints/DUSt3R_teacher.safetensors"
  eval_fp16: true

student:
  arch: "dust3r_student_s"
  student_config:
    encoder_layers: 10
    mha_heads_ratio: 0.8
    ffn_ratio: 0.8
  init_from_pruned: false

distill:
  enable: true
  kd_temperature: [3, 5]       # 使用第一个值: 3
  beta_kd: [0.5, 0.7]          # 使用第一个值: 0.5
  gamma_fd: [0.0, 0.1]
  curriculum_pct_task_first: 0.7

optim:
  lr: 2e-4
  weight_decay: 0.01
  sched: "cosine"
  grad_clip: 1.0

budget:
  gpu_hours_soft: 48
  stop_if_exceed: true
```
